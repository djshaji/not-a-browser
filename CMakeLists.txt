cmake_minimum_required(VERSION 3.16)
project(StreamGallery VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
set(RESOURCE_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/stream-gallery-resources.c")
add_custom_command(OUTPUT "${RESOURCE_SOURCE}"
    COMMAND "${GLIB_COMPILE_RESOURCES}" stream-gallery.gresource.xml
        --sourcedir=${CMAKE_CURRENT_SOURCE_DIR}/resources --generate-source
        --target=${RESOURCE_SOURCE}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources"
    DEPENDS resources/stream-gallery.gresource.xml resources/ui/main-window.ui resources/application.css resources/scripts/extract-hls.js examples/adapters/generic-hls.json
    VERBATIM)
add_library(stream-gallery-ui STATIC src/ui/MainWindow.cpp "${RESOURCE_SOURCE}")
target_include_directories(stream-gallery-ui PUBLIC src)
target_link_libraries(stream-gallery-ui PUBLIC PkgConfig::GTK3)
target_compile_options(stream-gallery-ui PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
add_executable(stream-gallery src/main.cpp src/app/Application.cpp)
target_link_libraries(stream-gallery PRIVATE stream-gallery-ui)
target_compile_options(stream-gallery PRIVATE -Wall -Wextra -Wpedantic)
option(BUILD_PLAYER "Build libmpv video playback" ON)
option(BUILD_LIBRARY "Build the SQLite library and command-line tool" ON)
if(BUILD_PLAYER)
    pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
    pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
    target_sources(stream-gallery-ui PRIVATE src/ui/PlayerView.cpp)
    target_link_libraries(stream-gallery-ui PRIVATE PkgConfig::MPV PkgConfig::EPOXY)
    target_compile_definitions(stream-gallery-ui PRIVATE HAVE_PLAYER)
endif()
option(BUILD_DISCOVERY "Build WebKit DOM-based HLS discovery" ON)
if(BUILD_DISCOVERY AND BUILD_PLAYER)
    pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1>=2.40)
    pkg_check_modules(DISCOVERY_SOUP REQUIRED IMPORTED_TARGET libsoup-3.0)
    target_sources(stream-gallery-ui PRIVATE src/ui/BrowserView.cpp src/services/HlsDiscovery.cpp)
    if(NOT BUILD_LIBRARY)
        target_sources(stream-gallery-ui PRIVATE src/services/ThumbnailService.cpp)
    endif()
    target_link_libraries(stream-gallery-ui PRIVATE PkgConfig::WEBKIT PkgConfig::DISCOVERY_SOUP)
    target_compile_definitions(stream-gallery-ui PRIVATE HAVE_DISCOVERY HAVE_SOUP)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME adapter-runner COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/adapter_runner_tests.py")
    endif()
    add_executable(ui-smoke-tests tests/ui_smoke_tests.cpp)
    target_link_libraries(ui-smoke-tests PRIVATE stream-gallery-ui)
    if(BUILD_PLAYER)
        target_compile_definitions(ui-smoke-tests PRIVATE HAVE_PLAYER_TESTS)
    endif()
    find_program(XVFB_RUN xvfb-run)
    if(XVFB_RUN)
        add_test(NAME ui-smoke COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:ui-smoke-tests>)
    else()
        add_test(NAME ui-smoke COMMAND ui-smoke-tests)
    endif()
    if(BUILD_DISCOVERY AND BUILD_PLAYER)
        add_executable(hls-discovery-tests tests/hls_discovery_tests.cpp src/services/HlsDiscovery.cpp)
        target_include_directories(hls-discovery-tests PRIVATE src)
        target_link_libraries(hls-discovery-tests PRIVATE PkgConfig::DISCOVERY_SOUP)
        target_compile_options(hls-discovery-tests PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME hls-discovery COMMAND hls-discovery-tests)
        set_tests_properties(hls-discovery PROPERTIES TIMEOUT 30 ENVIRONMENT "G_DEBUG=fatal-warnings")
        add_executable(discovery-tests tests/discovery_tests.cpp)
        target_link_libraries(discovery-tests PRIVATE stream-gallery-ui PkgConfig::WEBKIT)
        if(XVFB_RUN)
            add_test(NAME discovery COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:discovery-tests>)
        else()
            add_test(NAME discovery COMMAND discovery-tests)
        endif()
        set_tests_properties(discovery PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90
            ENVIRONMENT "G_DEBUG=fatal-warnings;GTK_THEME=Adwaita")
    endif()
    set_tests_properties(ui-smoke PROPERTIES SKIP_RETURN_CODE 77 ENVIRONMENT "G_DEBUG=fatal-warnings;GTK_THEME=Adwaita")
endif()
install(TARGETS stream-gallery RUNTIME DESTINATION bin)

if(BUILD_LIBRARY)
    find_package(SQLite3 REQUIRED)
    pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0>=2.66)
    add_library(stream-gallery-database src/database/Database.cpp src/database/LibraryRepository.cpp)
    target_include_directories(stream-gallery-database PUBLIC src)
    if(TARGET SQLite3::SQLite3)
        set(STREAM_GALLERY_SQLITE_TARGET SQLite3::SQLite3)
    else()
        set(STREAM_GALLERY_SQLITE_TARGET SQLite::SQLite3)
    endif()
    target_link_libraries(stream-gallery-database PUBLIC ${STREAM_GALLERY_SQLITE_TARGET} PkgConfig::GLIB)
    target_sources(stream-gallery-ui PRIVATE src/ui/GalleryView.cpp src/services/ThumbnailService.cpp)
    target_link_libraries(stream-gallery-ui PUBLIC stream-gallery-database)
    target_compile_definitions(stream-gallery-ui PRIVATE HAVE_LIBRARY)
    pkg_check_modules(SOUP QUIET IMPORTED_TARGET libsoup-3.0)
    if(SOUP_FOUND)
        target_link_libraries(stream-gallery-ui PRIVATE PkgConfig::SOUP)
        target_compile_definitions(stream-gallery-ui PRIVATE HAVE_SOUP)
    else()
        message(STATUS "libsoup-3.0 development files absent: thumbnails support file:// URLs and disk cache only")
    endif()
    target_compile_options(stream-gallery-database PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(stream-gallery-library src/database/LibraryTool.cpp)
    target_link_libraries(stream-gallery-library PRIVATE stream-gallery-database)
    install(TARGETS stream-gallery-library RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        target_compile_definitions(ui-smoke-tests PRIVATE HAVE_LIBRARY_TESTS)
        if(TARGET discovery-tests)
            target_compile_definitions(discovery-tests PRIVATE HAVE_LIBRARY_TESTS)
        endif()
        add_executable(thumbnail-tests tests/thumbnail_tests.cpp)
        target_link_libraries(thumbnail-tests PRIVATE stream-gallery-ui)
        add_test(NAME thumbnails COMMAND thumbnail-tests)
        set_tests_properties(thumbnails PROPERTIES SKIP_RETURN_CODE 77 ENVIRONMENT "G_DEBUG=fatal-warnings")
        add_executable(database-tests tests/database_tests.cpp)
        target_link_libraries(database-tests PRIVATE stream-gallery-database)
        add_test(NAME database COMMAND database-tests)
    endif()
endif()
